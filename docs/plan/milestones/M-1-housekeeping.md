```sh
#!/bin/sh
# house rules for commit messages: RULES.md Attribution and Git conventions
# usage: commit-msg <message-file>. CI runs it once per commit in the pushed range.
f="$1"
msg=$(grep -v '^#' "$f")
subj=$(printf '%s\n' "$msg" | sed -n '1p')
fail() { echo "commit-msg: $1" >&2; exit 1; }
printf '%s\n' "$msg" | grep -Eiq '^[[:space:]]*[a-z-]+-(by|with):' && fail "trailers are not allowed"
printf '%s\n' "$msg" | grep -Eiq 'co-authored|generated[- ](with|by)|noreply@anthropic|ai-generated' && fail "no ai or llm credit"
# a tool name and a credit word in either order within one sentence, or a
# credit key such as Author: naming a tool. Whole words only, so a scoped
# subject such as `settings: empty the claude attribution keys` passes.
ai='(^|[^[:alnum:]])(claude|anthropic|chatgpt|openai|copilot|gemini|llm|ai assistant)([^[:alnum:]]|$)'
credit='(^|[^[:alnum:]])(wrote|written|generated|assisted|authored|drafted|made with|created with|help|helped|thanks|courtesy of|powered by)([^[:alnum:]]|$)'
key='^[[:space:]]*(authors?|co-authors?|credits?|assistant|writer|thanks)[[:space:]]*:'
printf '%s\n' "$msg" | grep -Eiq "$ai[^.]*$credit|$credit[^.]*$ai|$key.*$ai" && fail "no ai or llm credit"
printf '%s\n' "$msg" | LC_ALL=C grep -q "$(printf '\342\200[\222-\225]')" && fail "no em or en dashes"
printf '%s\n' "$msg" | LC_ALL=C grep -q "$(printf '\342\210\222')" && fail "no minus sign as a dash"
printf '%s\n' "$msg" | LC_ALL=C grep -Eq "$(printf '\360\237')|$(printf '\342[\230-\236]')|$(printf '\357\270\217')" && fail "no emoji or dingbats"
[ -n "$subj" ] || fail "subject is empty"
[ ${#subj} -lt 60 ] || fail "subject is ${#subj} characters, keep it under 60"
printf '%s' "$subj" | grep -Eq '^[A-Z]' && fail "subject starts with a capital, start it lowercase"
printf '%s' "$subj" | grep -q '!' && fail "no exclamation marks"
printf '%s' "$subj" | grep -Eq '[.,;:]$' && fail "subject ends in punctuation, drop it"
printf '%s' "$subj" | grep -Eq '^(feat|fix|chore|docs|refactor|test|perf|build|ci|style|revert)(\(|!|:)' && fail "no type word in the subject"
printf '%s\n' "$msg" | grep -Eq '(^|[^-])-- ' && fail "no double hyphen as a dash"
if [ -f tools/lint-voice.mjs ] && command -v node >/dev/null 2>&1; then
  printf '%s\n' "$msg" | node tools/lint-voice.mjs --stdin --kind commit || fail "lint-voice rejected the message"
fi
exit 0
```
